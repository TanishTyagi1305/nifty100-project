import pandas as pd
import os
import glob

# ─────────────────────────────────────────
# STEP 1: Load all CSV files from data/raw
# ─────────────────────────────────────────

RAW_DATA_PATH = "data/raw"
csv_files = glob.glob(os.path.join(RAW_DATA_PATH, "*.csv"))

if not csv_files:
    print("⚠️  No CSV files found in data/raw/. Please place your CSV files there.")
else:
    print(f"✅ Found {len(csv_files)} CSV file(s)\n")

all_dataframes = {}

for filepath in csv_files:
    filename = os.path.basename(filepath)
    name = filename.replace(".csv", "")
    print(f"\n{'='*60}")
    print(f"📂 FILE: {filename}")
    print(f"{'='*60}")

    try:
        df = pd.read_csv(filepath)
        all_dataframes[name] = df

        print(f"\n📐 Shape (rows x columns): {df.shape}")
        print(f"\n🔬 Data Types:")
        print(df.dtypes.to_string())
        print(f"\n👀 First 5 Rows:")
        print(df.head().to_string())

        # ── Anomaly Detection ──
        print(f"\n⚠️  Anomaly Check:")
        missing = df.isnull().sum()
        if missing.any():
            print(f"  Missing values:\n{missing[missing > 0]}")
        else:
            print("  ✅ No missing values")

        dups = df.duplicated().sum()
        print(f"  Duplicate rows: {dups}")

    except Exception as e:
        print(f"  ❌ Error loading {filename}: {e}")

# ─────────────────────────────────────────────────────────
# STEP 2: Explore fund_master (if the file is present)
# ─────────────────────────────────────────────────────────
if "fund_master" in all_dataframes:
    print("\n\n" + "="*60)
    print("📊 FUND MASTER — Exploration")
    print("="*60)
    fm = all_dataframes["fund_master"]

    for col in ["fund_house", "category", "sub_category", "risk_grade"]:
        if col in fm.columns:
            print(f"\n🔹 Unique {col}s ({fm[col].nunique()}):")
            print(fm[col].unique())
        else:
            print(f"\n⚠️  Column '{col}' not found in fund_master")

# ─────────────────────────────────────────────────────────
# STEP 3: Validate AMFI codes
# ─────────────────────────────────────────────────────────
if "fund_master" in all_dataframes and "nav_history" in all_dataframes:
    print("\n\n" + "="*60)
    print("✅ AMFI CODE VALIDATION")
    print("="*60)
    fm = all_dataframes["fund_master"]
    nh = all_dataframes["nav_history"]

    # Adjust column names if yours are different
    master_codes = set(fm["scheme_code"].astype(str))
    nav_codes    = set(nh["scheme_code"].astype(str))

    missing_in_nav = master_codes - nav_codes
    extra_in_nav   = nav_codes - master_codes

    print(f"\nTotal schemes in fund_master : {len(master_codes)}")
    print(f"Total schemes in nav_history : {len(nav_codes)}")
    print(f"Codes in master but NOT in nav_history : {len(missing_in_nav)}")
    print(f"Codes in nav_history but NOT in master : {len(extra_in_nav)}")

    with open("reports/data_quality_summary.txt", "w") as f:
        f.write("DATA QUALITY SUMMARY\n")
        f.write("="*40 + "\n\n")
        f.write(f"Schemes in fund_master    : {len(master_codes)}\n")
        f.write(f"Schemes in nav_history    : {len(nav_codes)}\n")
        f.write(f"Missing from nav_history  : {len(missing_in_nav)}\n")
        f.write(f"Extra in nav_history      : {len(extra_in_nav)}\n\n")
        if missing_in_nav:
            f.write("Missing codes:\n")
            for c in sorted(missing_in_nav):
                f.write(f"  {c}\n")
    print("\n📄 Saved: reports/data_quality_summary.txt")
else:
    print("\n⚠️  Skipping AMFI validation — fund_master or nav_history not found.")
